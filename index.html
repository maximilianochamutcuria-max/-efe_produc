<script>


const base = "images";


const albums = [

  "gymna_vs_cariocas"

];


function cargarAlbumes() {

  const albumsDiv = document.getElementById("albums");


  albums.forEach(album => {

    const btn = document.createElement("button");

    btn.innerText = album;

    btn.onclick = () => cargarFotos(album);

    albumsDiv.appendChild(btn);

  });

}


async function cargarFotos(album) {

  const res = await fetch(`${base}/${album}/preview`);

  const text = await res.text();


  const parser = new DOMParser();

  const html = parser.parseFromString(text, "text/html");

  const links = html.querySelectorAll("a");


  const gallery = document.getElementById("gallery");

  gallery.innerHTML = "";


  links.forEach(link => {

    const file = link.getAttribute("href");


    if(file.endsWith(".jpg")) {

      const img = document.createElement("img");

      img.src = `${base}/${album}/preview/${file}`;

      img.onclick = () => abrirModal(img.src);

      gallery.appendChild(img);

    }

  });

}


function abrirModal(src) {

  document.getElementById("modal").style.display = "flex";

  document.getElementById("modalImg").src = src;

}


function cerrarModal() {

  document.getElementById("modal").style.display = "none";

}


cargarAlbumes();


</script>
